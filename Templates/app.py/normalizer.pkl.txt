# normalizer_creator.py

import pandas as pd
from sklearn.preprocessing import StandardScaler
import pickle

# Example dataset (replace with real liver health data)
# Columns: ALT, AST, ALP, Bilirubin, Albumin, Age
data = pd.DataFrame({
    "ALT": [45, 67, 38],
    "AST": [35, 55, 29],
    "ALP": [120, 133, 110],
    "Bilirubin": [0.8, 1.2, 0.9],
    "Albumin": [3.5, 3.8, 4.0],
    "Age": [52, 60, 47]
})

# Create and fit normalizer (standard scaler)
scaler = StandardScaler()
scaler.fit(data)

# Save to normalizer.pkl
with open("normalizer.pkl", "wb") as f:
    pickle.dump(scaler, f)

print("? normalizer.pkl created and saved.")
